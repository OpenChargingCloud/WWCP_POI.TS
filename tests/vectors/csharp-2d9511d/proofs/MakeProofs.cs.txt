// Makes charging station proofs (SPECIFICATION.md section 8.4) of the head of QA as the examples left it, attested by the
// signing keys of the three holders of OCC, as test vectors for the TypeScript port. Reads a copy of the history store.

using Org.BouncyCastle.Crypto;
using Org.BouncyCastle.Security;

using org.GraphDefined.Vanaheimr.Illias;

using cloud.charging.open.protocols.WWCP.POI;

var examples  = args[0];   // the directory of the C# examples, with AlphaOrg/, BetaOrg/, GammaOrg/
var state     = args[1];   // a copy of poi/QA and poi/QA-archive
var output    = args[2];   // where to write the proofs
var at        = DateTimeOffset.Parse(args[3]);

static AsymmetricKeyParameter ReadKey(String File)
{
    var pem   = System.IO.File.ReadAllText(File);
    var body  = String.Concat(pem.Split('\n').Where(line => !line.StartsWith("-----")).Select(line => line.Trim()));
    return PrivateKeyFactory.CreateKey(Convert.FromBase64String(body));
}

using var store    = new RoamingNetworkHistoryFileStore  (Path.Combine(state, "QA"));
using var archive  = new RoamingNetworkHistoryFileArchive(Path.Combine(state, "QA-archive"));

if (!RoamingNetworkHistory.TryLoad(store, out var history, out var error, Archive: archive))
    throw new Exception(error);

history.AttestationSigners = [ .. new[] { "AlphaOrg", "BetaOrg", "GammaOrg" }.Select(holder => {
                                    var key = ReadKey(Path.Combine(examples, holder, "signing-privateKey.pem"));
                                    return new RoamingNetworkCommitSigner(key,
                                                                          POIPublicKey.From(key).Id,
                                                                          COSEAlgorithm.ES256,
                                                                          RoamingNetworkCommitSignatureRoles.RoamingNetworkOperator,
                                                                          "OCC");
                               }) ];

if (!history.TryAttest(out var attestations, out error, At: at))
    throw new Exception(error);

Directory.CreateDirectory(output);

foreach (var attestation in attestations)
    Console.WriteLine($"attested revision {attestation.Revision}, {attestation.ETag}, by {attestation.KeyId[..8]}…");

File.WriteAllText(Path.Combine(output, "QA-attestations.json"), new Newtonsoft.Json.Linq.JArray(attestations.Select(attestation => attestation.ToJSON())).ToString());

foreach (var stationId in new[] { "DE*GEF*S1", "DE*GEF*S6" })
{

    if (!history.TryGetChargingStationProof(ChargingStation_Id.Parse(stationId), out var proof, out error, At: at))
        throw new Exception(error);

    var trustAnchorText = File.ReadAllText(Path.Combine(examples, "published", "QA-trustAnchor.txt"));

    if (!RoamingNetworkTrustAnchor.TryParse(trustAnchorText, out var anchor, out error))
        throw new Exception(error);

    Console.WriteLine($"proof of {stationId}: {(proof.Verify(anchor, out _, out error, at) ? "valid" : error)}");

    File.WriteAllText(Path.Combine(output, $"QA-proof-station-{stationId.Replace("*", "_")}.json"), proof.ToJSON().ToString());

}

if (!history.TryGetChargingStationProof(EVSE_Id.Parse("DE*GEF*E3*2"), out var evseProof, out error, At: at))
    throw new Exception(error);

File.WriteAllText(Path.Combine(output, "QA-proof-evse-DE_GEF_E3_2.json"), evseProof.ToJSON().ToString());

// The proof of the commit of the head, revision 12, in the first checkpoint whose commits include it.
if (!history.TryGetCommitProof(history.Head.Commit.Id, out var commitProof, out error))
    throw new Exception(error);

File.WriteAllText(Path.Combine(output, "QA-proof-revision-12.json"), commitProof.ToJSON().ToString());
Console.WriteLine($"proof of revision 12 in checkpoint {commitProof.CheckpointId}");


// ECDSA malleability: (r, s) and (r, n - s) are both valid ECDSA signatures of the same input. A price statement's digest
// covers its signature value, so the twin is another statement of the same sequence.
{

    var statements  = Newtonsoft.Json.Linq.JArray.Parse(File.ReadAllText(Path.Combine(examples, "published", "QA-checkpoint-4-priceStatements.json")));
    var original    = (Newtonsoft.Json.Linq.JObject) statements[0];

    using var reader = new Newtonsoft.Json.JsonTextReader(new StringReader(original.ToString())) { DateParseHandling = Newtonsoft.Json.DateParseHandling.None, FloatParseHandling = Newtonsoft.Json.FloatParseHandling.Decimal };
    var originalJSON = Newtonsoft.Json.Linq.JObject.Load(reader);

    var value  = Convert.FromBase64String(originalJSON["value"]!.ToString());
    var n      = Org.BouncyCastle.Asn1.Nist.NistNamedCurves.GetByName("P-256").N;
    var s      = new Org.BouncyCastle.Math.BigInteger(1, value, 32, 32);
    var twin   = (byte[]) value.Clone();
    Org.BouncyCastle.Utilities.BigIntegers.AsUnsignedByteArray(n.Subtract(s), twin, 32, 32);

    var twinJSON = (Newtonsoft.Json.Linq.JObject) originalJSON.DeepClone();
    twinJSON["value"] = Convert.ToBase64String(twin);

    if (!RoamingNetwork.TryParse(Helpers2.ReadJSON(Path.Combine(examples, "published", "QA.json")), out var qa, out error) ||
        !qa.TryGetChargingStationOperatorById(ChargingStationOperator_Id.Parse("DE*GEF"), out var gef))
        throw new Exception(error);

    foreach (var (name, json) in new[] { ("original", originalJSON), ("twin (r, n-s)", twinJSON) })
    {
        if (!RoamingNetworkPriceStatement.TryParse(json, out var statement, out error))
            throw new Exception(error);
        Console.WriteLine($"price statement {name}: signature {(statement.TryVerify(gef.SigningKeys, out var verifyError) ? "valid" : verifyError)}, sequence {statement.Sequence}, digest {statement.Digest}");
    }

    File.WriteAllText(Path.Combine(output, "QA-priceStatement-malleated-twin.json"), twinJSON.ToString());

    // The server holds the genuine statement 1 of DE*GEF*E1*1 as the current one. Its twin, which anybody can make from it:
    var currentArray = Newtonsoft.Json.Linq.JArray.Parse(File.ReadAllText(Path.Combine(examples, "published", "QA-priceStatements-DE_GEF_E1_1.json")));
    using var reader2 = new Newtonsoft.Json.JsonTextReader(new StringReader(currentArray[0].ToString())) { DateParseHandling = Newtonsoft.Json.DateParseHandling.None, FloatParseHandling = Newtonsoft.Json.FloatParseHandling.Decimal };
    var genuine      = Newtonsoft.Json.Linq.JObject.Load(reader2);
    var genuineValue = Convert.FromBase64String(genuine["value"]!.ToString());
    var twinValue    = (byte[]) genuineValue.Clone();
    Org.BouncyCastle.Utilities.BigIntegers.AsUnsignedByteArray(n.Subtract(new Org.BouncyCastle.Math.BigInteger(1, genuineValue, 32, 32)), twinValue, 32, 32);
    var twin1        = (Newtonsoft.Json.Linq.JObject) genuine.DeepClone();
    twin1["value"]   = Convert.ToBase64String(twinValue);

    foreach (var (name, json) in new[] { ("genuine statement 1 again", genuine), ("twin of statement 1", twin1) })
    {
        if (!RoamingNetworkPriceStatement.TryParse(json, out var statement, out error))
            throw new Exception(error);
        var result = history.TryAddPriceStatement(statement, out var takeError, at);
        Console.WriteLine($"server takes the {name}: {result}, {takeError}");
    }

}

Console.WriteLine("done");


static class Helpers2
{
    public static Newtonsoft.Json.Linq.JObject ReadJSON(String File)
    {
        using var reader = new Newtonsoft.Json.JsonTextReader(new StringReader(System.IO.File.ReadAllText(File))) { DateParseHandling = Newtonsoft.Json.DateParseHandling.None, FloatParseHandling = Newtonsoft.Json.FloatParseHandling.Decimal };
        return Newtonsoft.Json.Linq.JObject.Load(reader);
    }
}
